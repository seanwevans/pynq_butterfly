cmake_minimum_required(VERSION 3.16.3)
project(pynq_butterfly_openfhe LANGUAGES CXX)
option(PYNQ_OPENFHE_BUILD_TOOLS "Build command-line tools that require OpenFHE" ON)
include(CTest)
add_subdirectory(lib)
if(PYNQ_OPENFHE_BUILD_TOOLS)
  find_package(OpenFHE CONFIG REQUIRED)
  set(_entries ${OpenFHE_CXX_FLAGS})
  set(PYNQ_OPENFHE_FLAGS)
  foreach(entry IN LISTS _entries)
    separate_arguments(parts UNIX_COMMAND "${entry}")
    foreach(flag IN LISTS parts)
      if(NOT flag MATCHES "^-W" AND NOT flag STREQUAL "-pedantic" AND NOT flag STREQUAL "-pedantic-errors")
        list(APPEND PYNQ_OPENFHE_FLAGS "${flag}")
      endif()
    endforeach()
  endforeach()
  add_subdirectory(tools)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
